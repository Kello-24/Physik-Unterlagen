\documentclass[14pt,aspectratio=169]{beamer}
\input{gemeinsam-praeambel}

\setbeamercolor{frametitle}{fg=kfrblau,bg=white}

\setbeamertemplate{frametitle}{%
  \nointerlineskip
  \vspace{0.25cm}%
  \begin{beamercolorbox}[wd=\paperwidth,sep=0.2em,leftskip=0.5cm,rightskip=2.7cm plus 1fil]{frametitle}
    \usebeamerfont{frametitle}\insertframetitle
  \end{beamercolorbox}%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=north east,inner sep=0pt] at ([xshift=-0.35cm,yshift=-0.25cm]current page.north east)
      {\includegraphics[height=0.62cm]{kfr_logo}};
  \end{tikzpicture}%
  \par\nointerlineskip
  \makebox[0pt][l]{\color{kfrblau}\rule{\paperwidth}{1.5pt}}
}

\setbeamertemplate{footline}{%
  \hfill%
  \usebeamercolor[fg]{page number in head/foot}%
  \usebeamerfont{page number in head/foot}%
  \insertframenumber\,/\,\inserttotalframenumber\hspace{1em}\vspace{0.5em}
}
