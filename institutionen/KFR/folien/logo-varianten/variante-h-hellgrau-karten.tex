      {\usebeamerfont{frametitle}\strut\insertframetitle};

  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=north east,inner sep=0pt] at ([xshift=-0.5cm,yshift=-0.35cm]current page.north east)
      {\includegraphics[height=0.62cm]{kfr_logo}};
  \end{tikzpicture}%
}
