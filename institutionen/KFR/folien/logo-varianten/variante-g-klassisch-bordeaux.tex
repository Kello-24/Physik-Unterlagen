\definecolor{bordeaux}{HTML}{7A1F35}
\definecolor{hellbordeaux}{HTML}{F8EEF0}

\setbeamercolor{structure}{fg=bordeaux}

\setbeamerfont{frametitle}{size=\Large,series=\mdseries,shape=\scshape}

\setbeamertemplate{frametitle}{%
  \nointerlineskip
  \vspace*{0.3cm}%
  \begin{beamercolorbox}[wd=\paperwidth,center,leftskip=1cm,rightskip=1cm]{frametitle}

    {\color{bordeaux!60}\rule{0.5\paperwidth}{0.5pt}}\par
  \end{beamercolorbox}
}

  \begin{beamercolorbox}[wd=\paperwidth,ht=0.75cm,dp=0.2cm,leftskip=0.6cm,rightskip=0.6cm]{footline}

}
