\begin{frame}{Was wir schon kennen: Differenzen}
  \small
  Das Zeichen $\Delta$ (Delta) bedeutet immer: \textbf{Ende minus Anfang}.
  \[
    \Delta s = s_2 - s_1 \quad\text{(Ortsänderung)}, \qquad
    \Delta t = t_2 - t_1 \quad\text{(vergangene Zeit)}
  \]
  Damit haben wir zwei Grössen definiert:
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{merkbox}[Geschwindigkeit]
        \[
          v = \frac{\Delta s}{\Delta t}
        \]
        Einheit: $\si{\meter\per\second}$
      \end{merkbox}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{merkbox}[Beschleunigung]
        \[
          a = \frac{\Delta v}{\Delta t}
        \]
        Einheit: $\si{\meter\per\second\squared}$
      \end{merkbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Mit Einheiten rechnen}
  \small
  Einheiten rechnen wir wie Zahlen mit: Wir multiplizieren, dividieren und
  kürzen sie. So sieht man am Schluss, ob das Resultat die richtige
  Einheit hat.
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Malnehmen und Kürzen}
      \begin{align*}
        \SI{5}{\meter\per\second}\cdot\SI{240}{\second}
          &= 5\cdot240\cdot\frac{\ein{m}}{\kue{s}}\cdot\kue{s}
           = \SI{1200}{\meter}\\
        \SI{2}{\meter\per\second\squared}\cdot\SI{5}{\second}
          &= 2\cdot5\cdot\frac{\ein{m}}{\ein{s}\cdot\kue{s}}\cdot\kue{s}
           = \SI{10}{\meter\per\second}
      \end{align*}
      \textbf{Addieren} nur bei gleicher Einheit:
      \[
        \SI{12}{\meter}+\SI{18}{\meter}=\SI{30}{\meter}
      \]
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{Durch einen Bruch teilen}\\
      heisst: mit dem Kehrwert malnehmen.
      \begin{align*}
        \frac{\SI{1000}{\meter}}{\SI{5}{\meter\per\second}}
          &= \frac{1000}{5}\cdot\ein{m}\cdot\frac{\ein{s}}{\ein{m}}\\
          &= 200\cdot\frac{\kue{m}\cdot\ein{s}}{\kue{m}}
           = \SI{200}{\second}
      \end{align*}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Aufgabe 1: Lösung a)}
  \small
  a) Steigung aus zwei Punkten der Geraden: Anfang
  $(\SI{0}{\second}\,|\,\SI{2}{\meter\per\second})$, Ende
  $(\SI{6}{\second}\,|\,\SI{8}{\meter\per\second})$.
  \begin{align*}
    a &= \frac{\Delta v}{\Delta t}
       = \frac{\SI{8}{\meter\per\second}-\SI{2}{\meter\per\second}}{\SI{6}{\second}-\SI{0}{\second}}
      && \Eh{Ende minus Anfang}\\
      &= \frac{\SI{6}{\meter\per\second}}{\SI{6}{\second}}
      && \Eh{gleiche Einheiten: einfach subtrahieren}\\
      &= \frac{6}{6}\cdot\frac{\;\frac{\ein{m}}{\ein{s}}\;}{\ein{s}}
      && \Eh{Zahlen und Einheiten getrennt}\\
      &= 1\cdot\frac{\ein{m}}{\ein{s}\cdot\ein{s}}
       = \SI{1}{\meter\per\second\squared}
      && \Eh{durch s teilen: ein zweites s in den Nenner}
  \end{align*}
\end{frame}

      \[
        \SI{2}{\meter\per\second}\cdot\SI{6}{\second}
        = 2\cdot6\cdot\frac{\ein{m}}{\kue{s}}\cdot\kue{s}
        = \SI{12}{\meter}
      \]
      \textbf{Dreieck:} Breite $\SI{6}{\second}$, Höhe

      \[
        \tfrac12\cdot\SI{6}{\second}\cdot\SI{6}{\meter\per\second}
        = \tfrac12\cdot6\cdot6\cdot\kue{s}\cdot\frac{\ein{m}}{\kue{s}}
        = \SI{18}{\meter}
      \]
      \textbf{Zusammen} (gleiche Einheit, also addieren):
      \[
        \Delta s=\SI{12}{\meter}+\SI{18}{\meter}=\SI{30}{\meter}
      \]
    \end{column}
  \end{columns}

\begin{frame}{Das Wichtigste: $t-t_0$ ist einfach $\Delta t$}
  \small
  $t$ und $t_0$ sind \textbf{Zeitpunkte} (wie eine Uhrzeit).
  $t-t_0$ ist die \textbf{Zeitdauer} dazwischen: die Zeit, die seit dem
  Start vergangen ist.

  \centering
  \begin{tikzpicture}[scale=0.95]
    \draw[->,thick] (-0.3,0) -- (12.5,0) node[right] {Uhrzeit};
    \foreach \x/\l in {0/18,2/20,4/22,6/24,8/26,10/28}

\begin{frame}{Aufgabe 2: Start und später}

      \item[b)] Berechne $t-t_0$ (in Sekunden) und $s-s_0$ (in Metern).
      \item[c)] Berechne Laras Geschwindigkeit $v$ in

    \end{enumerate}

\begin{frame}{Aufgabe 2: Lösung a) und b)}
  \small
  a) $t_0=7^{20}$ Uhr, \ $s_0=\SI{1.5}{\kilo\meter}$ \ (Bahnhof);
  \quad $t=7^{26}$ Uhr, \ $s=\SI{3.3}{\kilo\meter}$ \ (Kanti)

  \vspace{0.5em}
  b)
  \begin{align*}
    t-t_0 &= 7^{26}-7^{20} = \SI{6}{\minute}
      && \Eh{Zeitdauer zwischen zwei Uhrzeiten}\\
          &= 6\cdot\SI{60}{\second} = \SI{360}{\second}
      && \Eh{$\SI{1}{\minute}=\SI{60}{\second}$}\\[0.4em]
    s-s_0 &= \SI{3.3}{\kilo\meter}-\SI{1.5}{\kilo\meter} = \SI{1.8}{\kilo\meter}
      && \Eh{gleiche Einheit: einfach subtrahieren}\\
          &= 1.8\cdot\SI{1000}{\meter} = \SI{1800}{\meter}
      && \Eh{$\SI{1}{\kilo\meter}=\SI{1000}{\meter}$}
  \end{align*}

\begin{frame}{Aufgabe 2: Lösung c)}
  \small
  c) Geschwindigkeit in $\si{\meter\per\second}$:
  \begin{align*}
    v &= \frac{s-s_0}{t-t_0}
       = \frac{\SI{1800}{\meter}}{\SI{360}{\second}}
       = \frac{1800}{360}\cdot\frac{\ein{m}}{\ein{s}}
       = \SI{5}{\meter\per\second}
      && \Eh{m geteilt durch s ergibt m/s}
  \end{align*}
  Umrechnen in $\si{\kilo\meter\per\hour}$, mit
  $\SI{1}{\meter}=\tfrac{1}{1000}\,\si{\kilo\meter}$ und
  $\SI{1}{\second}=\tfrac{1}{3600}\,\si{\hour}$:
  \begin{align*}
    \SI{5}{\meter\per\second}
      &= 5\cdot\frac{\tfrac{1}{1000}\,\ein{km}}{\tfrac{1}{3600}\,\ein{h}}
      && \Eh{m und s durch km und h ersetzen}\\
      &= 5\cdot\frac{3600}{1000}\cdot\frac{\ein{km}}{\ein{h}}
       = 5\cdot3.6\cdot\frac{\ein{km}}{\ein{h}}
      && \Eh{daher kommt der Faktor 3.6}\\
      &= \SI{18}{\kilo\meter\per\hour}
  \end{align*}
\end{frame}

\begin{frame}{Aufgabe 4: Gleichung 2 anwenden}
  \begin{aufgabebox}[Aufgabe 4]

    Stoppuhr läuft schon. Bei $t_0=\SI{3}{\second}$ hat das Auto

\begin{frame}{Aufgabe 4: Lösung a)}
  \small

  \begin{align*}

      && \Eh{Werte mit Einheiten einsetzen}\\
      &= \SI{15}{\meter\per\second}+2\cdot5\cdot\frac{\ein{m}}{\ein{s}\cdot\kue{s}}\cdot\kue{s}
      && \Eh{$\si{\second\squared}=\ein{s}\cdot\ein{s}$, ein s kürzt sich}\\
      &= \SI{15}{\meter\per\second}+\SI{10}{\meter\per\second}
       = \SI{25}{\meter\per\second}
      && \Eh{gleiche Einheit: einfach addieren}\\
      &= 25\cdot3.6\,\si{\kilo\meter\per\hour}
       = \SI{90}{\kilo\meter\per\hour}
      && \Eh{m/s mal 3.6 ergibt km/h}
  \end{align*}

\begin{frame}{Aufgabe 4: Lösung b)}
  \small
  b) Gesucht ist zuerst die Zeitdauer $t-t_0$:
  \begin{align*}
    \SI{27}{\meter\per\second} &= \SI{15}{\meter\per\second}+\SI{2}{\meter\per\second\squared}\cdot(t-t_0)
      && \Eh{$-\SI{15}{\meter\per\second}$ auf beiden Seiten}\\
    \SI{12}{\meter\per\second} &= \SI{2}{\meter\per\second\squared}\cdot(t-t_0)
      && \Eh{$:\SI{2}{\meter\per\second\squared}$ auf beiden Seiten}\\
    t-t_0 &= \frac{\SI{12}{\meter\per\second}}{\SI{2}{\meter\per\second\squared}}

           = \SI{6}{\second}

\begin{frame}{Gleichung 4: Ohne die Zeit}
  \small

  Bremsweg. Dann kann man $\Tdiff$ loswerden:
  \begin{enumerate}
    \item Gleichung 2 nach der Zeitdauer auflösen:
      $\Tdiff=\dfrac{v-v_0}{a}$
    \item Das in Gleichung 3 einsetzen und vereinfachen (alle Schritte
      stehen auf dem Arbeitsblatt).
  \end{enumerate}
  \vspace{0.2em}
  \begin{merkbox}[Gleichung 4 ($a$ konstant, ohne Zeit)]
    \centering
    $v^2=\Vzero^2+2\,\Acol\,\Delta s$ \qquad mit $\Delta s=s-\Szero$
  \end{merkbox}
  \footnotesize
  Achtung Vorzeichen: Wird ein Objekt langsamer, zeigt $a$ gegen die
  Bewegungsrichtung. Ist die Fahrtrichtung positiv, ist $a$ dann negativ.
\end{frame}

\begin{frame}{Aufgabe 6: Gleichung 4 anwenden}
  \begin{aufgabebox}[Aufgabe 6]
    \small
    Ein Velofahrer fährt mit $\SI{8}{\meter\per\second}$ und bremst dann
    gleichmässig, bis er steht. Die Bremsverzögerung beträgt
    $\SI{2}{\meter\per\second\squared}$. Die Fahrtrichtung ist positiv.
    \begin{enumerate}
      \item[a)] Welches Vorzeichen hat $a$? Begründe.
      \item[b)] Wie lang ist der Bremsweg $\Delta s$?
      \item[c)] Wie lange dauert das Bremsen, also wie gross ist $t-t_0$?
        Welche Gleichung passt hier?
    \end{enumerate}
  \end{aufgabebox}
\end{frame}

\begin{frame}{Aufgabe 6: Lösung a)}
  \small
  a) $a=-\SI{2}{\meter\per\second\squared}$.

  \vspace{0.4em}
  Das Velo fährt in die positive Richtung ($v>0$) und wird langsamer.
  Dann zeigt $a$ gegen die Fahrtrichtung, also in die negative Richtung.

  \vspace{0.4em}
  \begin{merkbox}[Merke]
    $v$ und $a$ mit verschiedenen Vorzeichen: langsamer werden.\\
    $v$ und $a$ mit gleichem Vorzeichen: schneller werden.
  \end{merkbox}
\end{frame}

\begin{frame}{Aufgabe 6: Lösung b)}
  \small
  b) Am Ende steht das Velo, $v=0$. Die Zeit kommt nicht vor: Gleichung 4.
  \begin{align*}
    v^2 &= v_0^2+2\,a\,\Delta s\\
    0 &= \left(\SI{8}{\meter\per\second}\right)^2+2\cdot\left(-\SI{2}{\meter\per\second\squared}\right)\cdot\Delta s
      && \Eh{Werte mit Einheiten einsetzen}\\
    0 &= \SI{64}{\meter\squared\per\second\squared}-\SI{4}{\meter\per\second\squared}\cdot\Delta s
      && \Eh{$\left(8\,\tfrac{\ein{m}}{\ein{s}}\right)^2=8^2\cdot\tfrac{\ein{m}^2}{\ein{s}^2}$: Zahl und Einheit quadrieren}\\
    \SI{4}{\meter\per\second\squared}\cdot\Delta s &= \SI{64}{\meter\squared\per\second\squared}
      && \Eh{$+\SI{4}{\meter\per\second\squared}\cdot\Delta s$ auf beiden Seiten}
  \end{align*}
\end{frame}

\begin{frame}{Aufgabe 6: Lösung b), Teil 2}
  \small
  Jetzt nach $\Delta s$ auflösen:
  \begin{align*}
    \SI{4}{\meter\per\second\squared}\cdot\Delta s &= \SI{64}{\meter\squared\per\second\squared}
      && \Eh{von der letzten Folie}\\
    \Delta s &= \frac{\SI{64}{\meter\squared\per\second\squared}}{\SI{4}{\meter\per\second\squared}}
              = 16\cdot\frac{\ein{m}^2}{\ein{s}^2}\cdot\frac{\ein{s}^2}{\ein{m}}
      && \Eh{$:\SI{4}{\meter\per\second\squared}$, also mal s$^2$/m}\\
             &= 16\cdot\frac{\ein{m}\cdot\kue{m}\cdot\kue{\ein{s}^2}}{\kue{\ein{s}^2}\cdot\kue{m}}
              = \SI{16}{\meter}
      && \Eh{s$^2$ und ein m kürzen sich, es bleibt m}
  \end{align*}
\end{frame}

\begin{frame}{Aufgabe 6: Lösung c)}
  \small
  c) Gesucht ist eine Zeitdauer, der Ort spielt keine Rolle: also
  Gleichung 2.
  \begin{align*}
    v &= v_0+a\cdot(t-t_0)\\
    0 &= \SI{8}{\meter\per\second}+\left(-\SI{2}{\meter\per\second\squared}\right)\cdot(t-t_0)
      && \Eh{Werte mit Einheiten einsetzen}\\
    0 &= \SI{8}{\meter\per\second}-\SI{2}{\meter\per\second\squared}\cdot(t-t_0)
      && \Eh{$+\SI{2}{\meter\per\second\squared}\cdot(t-t_0)$ auf beiden Seiten}\\
    \SI{2}{\meter\per\second\squared}\cdot(t-t_0) &= \SI{8}{\meter\per\second}
      && \Eh{$:\SI{2}{\meter\per\second\squared}$ auf beiden Seiten}\\
    t-t_0 &= \frac{\SI{8}{\meter\per\second}}{\SI{2}{\meter\per\second\squared}}
           = 4\cdot\frac{\kue{m}\cdot\kue{s}\cdot\ein{s}}{\kue{s}\cdot\kue{m}}
           = \SI{4}{\second}
      && \Eh{wie in Aufgabe 4b: es bleibt s}
  \end{align*}
  Das Velo braucht $\SI{4}{\second}$ bis zum Stillstand.
\end{frame}

% --- Zusammenfassung ----------------------------------------------------------
\begin{frame}{Zusammenfassung}
  \small
  \centering
  \begin{tabular}{lll}
    \toprule
    & \textbf{Gleichung} & \textbf{Woher?} \\
    \midrule
    1 & $s(t)=s_0+v\,(t-t_0)$ & $v=\Delta s/\Delta t$ \ ($a=0$) \\[0.2em]
    2 & $v(t)=v_0+a\,(t-t_0)$ & $a=\Delta v/\Delta t$ \\[0.2em]
    3 & $s(t)=s_0+v_0(t-t_0)+\tfrac12\,a\,(t-t_0)^2$ & Fläche: Rechteck $+$ Dreieck \\[0.2em]
    4 & $v^2=v_0^2+2\,a\,\Delta s$ & Gl.\ 2 und 3 ohne Zeit \\
    \bottomrule
  \end{tabular}

  \vspace{0.6em}
  \begin{merkbox}[Die drei wichtigsten Übersetzungen]
    \centering
    $t-t_0=\Delta t$ \ (Zeit seit dem Start) \qquad
    $s-s_0=\Delta s$ \qquad
    $v-v_0=\Delta v$
  \end{merkbox}
\end{frame}
