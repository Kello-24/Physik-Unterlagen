\begin{formelbox}[Formelsammlung]
\begin{align*}
  F_{\text{mag}} &= I\cdot l\cdot B\cdot\sin\alpha
    &&\text{magnetische Kraft, allgemein}\\
  F_{\text{mag}} &= I\cdot l\cdot B &&\text{Spezialfall } \alpha=90^\circ\\
  F_L &= q\cdot v\cdot B\cdot\sin\alpha &&\text{Lorentzkraft, allgemein}\\
  F_L &= q\cdot v\cdot B &&\text{Spezialfall } \alpha=90^\circ\\
  B &= \frac{\mu_0}{2\pi}\cdot\frac{I}{r} &&\text{Feld eines geraden Leiters, Abstand } r
\end{align*}
$\mu_0=4\pi\cdot10^{-7}\,\si{\newton\per\ampere\squared}=1{,}26\cdot10^{-6}\,\si{\newton\per\ampere\squared}$;
$I$ in \si{\ampere}; $l,\,r$ in \si{\meter}; $B$ in \si{\tesla}; $q$ in
\si{\coulomb}; $v$ in \si{\meter\per\second}; $\alpha$ = Winkel zwischen
Leiter/Geschwindigkeit und Feldlinien.
\end{formelbox}

% Aufgabe 1: Kraft zwischen den Leitungen einer Gleichspannungs-Freileitung
% Quelle: literatur/Kraft zwischen den Leitungen einer
%   Gleichspannungs-Freileitung _ LEIFIphysik.pdf (Schwierigkeitsgrad dort:
%   leichte Aufgabe). BILD: Abb. 1 dieser Quelle (Foto der FennoSkan-HGÜ-
%   Freileitung, Rauma/Finnland) -- rein motivierend/dekorativ, für die
%   Lösung nicht nötig (alle Werte stehen im Aufgabentext). Gegenüber der
%   Quelle angepasst: Die Quelle berechnet den Strom zunächst aus
%   Leistung/Spannung ($I=P/U$) -- diese Formel ist nicht Teil des hier
%   behandelten Stoffs, daher wird $I$ direkt gegeben (Wert unverändert
%   übernommen). Die abschliessende Frage "Anziehung oder Abstossung?" der
%   Quelle wurde weggelassen, da das bereits in Experiment 2/3 des
%   Arbeitsblatts an genau diesem Fall (entgegengesetzte Ströme) behandelt
%   wurde und hier nur wiederholen, nicht ergänzen würde.
% =============================================================================
\begin{questions}
\begin{taskbox}[Aufgabe 1: Kraft zwischen den Leitungen einer Gleichspannungsfreileitung]
\Schwierigkeit{leicht}
\question Eine Hochspannungs-Gleichstrom-Freileitung führt einen Strom
von $I=\SI{522}{\ampere}$. Auf einem $l=\SI{100}{\meter}$ langen Abschnitt
verlaufen zwei ihrer Leitungen im Abstand $r=\SI{0.80}{\meter}$ parallel
zueinander, wobei der Strom in den beiden Leitungen in
\textbf{entgegengesetzte} Richtungen fliesst.
\begin{center}
  % BILD: Bilder/hguue-freileitung.jpg (o.ä.) -- siehe Kommentar oben.
\end{center}
\begin{parts}
  \part[1] Berechnen Sie die magnetische Flussdichte $B$, die eine der
  beiden Leitungen am Ort der jeweils anderen erzeugt.
  \Antwortraum{1}
  \begin{solution}
  \[
    B = \frac{\mu_0}{2\pi}\cdot\frac{I}{r}
      = \frac{4\pi\cdot10^{-7}\,\si{\newton\per\ampere\squared}}{2\pi}\cdot\frac{\SI{522}{\ampere}}{\SI{0.80}{\meter}}
  \]
  \[
    B = 2\cdot10^{-7}\,\si{\newton\per\ampere\squared}\cdot\frac{\SI{522}{\ampere}}{\SI{0.80}{\meter}}
      \approx \SI{1.3e-4}{\tesla} = \SI{130}{\micro\tesla}
  \]
  \end{solution}
  \part[1] Berechnen Sie die magnetische Kraft auf den betrachteten
  $\SI{100}{\meter}$ langen Leitungsabschnitt.
  \Antwortraum{1}
  \begin{solution}
  Die beiden Leitungen stehen senkrecht zum jeweils anderen Feld
  ($\alpha=90^\circ$), also:
  \[
    F_{\text{mag}} = I\cdot l\cdot B = \SI{522}{\ampere}\cdot\SI{100}{\meter}\cdot\SI{1.3e-4}{\tesla} \approx \SI{6.8}{\newton}
  \]
  \end{solution}
\end{parts}
\end{taskbox}
\end{questions}

\begin{questions}
\begin{taskbox}[Aufgabe 2: Kraft auf einen Leiterrahmen]
\Schwierigkeit{mittelschwer}
\question Ein rechteckiger Leiterrahmen trägt einen Strom von
$I=\SI{5.0}{\ampere}$. Er hängt teilweise in einem begrenzten homogenen
Magnetfeld $B=\SI{0.20}{\tesla}$, das senkrecht aus der Zeichenebene
herauszeigt (Draufsicht). Die Feldgrenze verläuft waagrecht; der Rahmen
taucht von oben senkrecht in das Feld ein.
\begin{center}
\begin{minipage}[t]{6.5cm}\centering
\begin{tikzpicture}[>=Latex,scale=0.55]
  \draw[dashed,thick] (-2.3,0) -- (2.3,0);
  \node[font=\scriptsize] at (0,0.4) {Feldgrenze};
  \foreach \x in {-1.8,-0.9,0,0.9,1.8}{
    \foreach \y in {-0.7,-1.5,-2.3}{
      \node[font=\small,gray] at (\x,\y) {$\odot$};
    }
  }
  \draw[very thick] (-1,1.6) -- (-1,-1.9) -- (1,-1.9) -- (1,1.6);
  \draw[thick,-{Latex[length=2.5mm]}] (0.3,-1.9) -- (-0.3,-1.9);
  \node[font=\scriptsize] at (0,-2.6) {$l_1$};
\end{tikzpicture}\\
{\scriptsize (a) teilweise eingetaucht}
\end{minipage}\hfill
\begin{minipage}[t]{6.5cm}\centering
\begin{tikzpicture}[>=Latex,scale=0.55]
  \draw[dashed,thick] (-2.3,1.7) -- (2.3,1.7);
  \node[font=\scriptsize] at (0,2.1) {Feldgrenze};
  \foreach \x in {-1.8,-0.9,0,0.9,1.8}{
    \foreach \y in {1.0,0.2,-0.6,-1.4,-2.2}{
      \node[font=\small,gray] at (\x,\y) {$\odot$};
    }
  }
  \draw[very thick] (-1.4,1.2) -- (-1.4,-1.9) -- (1.4,-1.9) -- (1.4,1.2) -- cycle;
  \draw[thick,-{Latex[length=2.5mm]}] (0.3,-1.9) -- (-0.3,-1.9);
  \node[font=\scriptsize] at (0,-2.6) {$l_2$};
\end{tikzpicture}\\
{\scriptsize (b) vollständig eingetaucht}
\end{minipage}
\end{center}
{\scriptsize $\odot$ = Magnetfeld aus der Zeichenebene heraus (Draufsicht).
Der Pfeil an der unteren Kante zeigt die dortige Stromrichtung.}
\begin{parts}
  \part[2] In einer ersten Situation taucht der Rahmen so weit ein, dass
  nur seine untere, waagrechte Kante der Breite $l_1=\SI{10}{\centi\meter}$
  vollständig im Feld liegt; die beiden seitlichen (senkrechten) Kanten
  ragen mit gleich langen Stücken ins Feld hinein. Bestimmen Sie mit der
  Drei-Finger-Regel die Kraftrichtung auf jede der beiden seitlichen
  Kanten. Was bedeutet das für ihre gemeinsame Wirkung auf den Rahmen?
  Berechnen Sie anschliessend die resultierende Kraft auf den gesamten
  Rahmen.
  \Antwortraum{2}
  \begin{solution}
  In den beiden seitlichen Kanten fliesst der Strom in \textbf{entgegen\-gesetzte}
  Richtungen (der Rahmen ist geschlossen), aber beide gleich langen
  Teilstücke liegen im selben Feld $B$. Nach der Drei-Finger-Regel zeigen
  die beiden Kräfte deshalb in entgegengesetzte Richtungen. Da Feld und
  Länge des eingetauchten Stücks bei beiden Kanten gleich sind, sind auch
  die Kräfte gleich gross. Sie heben sich also gegenseitig auf. Nur die
  untere Kante trägt netto bei:
  \[
    F_{\text{mag}} = I\cdot l_1\cdot B = \SI{5.0}{\ampere}\cdot\SI{0.10}{\meter}\cdot\SI{0.20}{\tesla} = \SI{0.10}{\newton}
  \]
  Die Kraft zieht den Rahmen weiter ins Feld hinein.
  \end{solution}
  \part[2] In einer zweiten Situation ist der Rahmen so weit eingetaucht,
  dass er mit seiner vollen Breite $l_2=\SI{20}{\centi\meter}$ und all
  seinen Kanten vollständig im homogenen Feld liegt. Begründen Sie, wie
  gross die resultierende Kraft auf den Rahmen jetzt ist.
  \Antwortraum{2}
  \begin{solution}
  Jetzt liegen alle vier Kanten vollständig im selben homogenen Feld
  $B$. Für jedes der beiden gegenüberliegenden Kantenpaare (oben/unten
  und links/rechts) fliesst der Strom in entgegengesetzte Richtungen
  durch dasselbe Feld, die Kräfte auf jedes Paar heben sich also
  gegenseitig auf:
  \[
    F_{\text{gesamt}} = 0
  \]
  Ein vollständig in einem homogenen Feld liegender, stromdurchflossener
  Rahmen erfährt keine resultierende Kraft, sondern höchstens ein
  Drehmoment (falls er nicht parallel zum Feld ausgerichtet ist).
  \end{solution}
\end{parts}
\end{taskbox}
\end{questions}

% Aufgabe 3: Bestimmung der magnetischen Kraft (Formelumstellung)
% Quelle: literatur/Bestimmung der magnetischen Kraft - Formelumstellung _
%   LEIFIphysik.pdf (Schwierigkeitsgrad dort: leichte Aufgabe). Die Quelle
%   hat fuenf Teilaufgaben (a-e, je eine Groesse gesucht); davon bewusst
%   nur zwei uebernommen (a: F direkt berechnen: b: hier ist die
%   Quell-Teilaufgabe e, nach dem Winkel alpha aufloesen/arcsin), um
%   Wiederholung zu vermeiden -- die drei ausgelassenen Teilaufgaben (nach
%   I, l bzw. B aufloesen) trainieren dieselbe Umstellung nur mit anderen
%   Variablen. Aufgabentext und Zahlen von dort uebernommen, Loesungswege
%   selbst nachvollzogen.
% =============================================================================
\begin{questions}
\begin{taskbox}[Aufgabe 3: Bestimmung der magnetischen Kraft (Formelumstellung)]
\Schwierigkeit{leicht}
\question Lösen Sie die Formel $F_{\text{mag}}=I\cdot l\cdot B\cdot\sin\alpha$
jeweils nach der gesuchten Grösse auf und berechnen Sie diese.
\begin{parts}
  \part[2] Ein $l=\SI{5.0}{\centi\meter}$ langer Leiter, der von einem
  Strom der Stärke $I=\SI{1.5}{\ampere}$ durchflossen wird, liegt in einem
  Winkel der Weite $\alpha=45^\circ$ zu einem homogenen Magnetfeld mit
  einer Flussdichte von $B=\SI{0.10}{\tesla}$. Berechnen Sie den Betrag
  der magnetischen Kraft auf den Leiter.
  \Antwortraum{2}
  \begin{solution}
  \[
    F_{\text{mag}} = I\cdot l\cdot B\cdot\sin\alpha
      = \SI{1.5}{\ampere}\cdot\SI{5.0e-2}{\meter}\cdot\SI{0.10}{\tesla}\cdot\sin(45^\circ)
      \approx \SI{5.3e-3}{\newton} = \SI{5.3}{\milli\newton}
  \]
  \end{solution}
  \part[2] Ein gerader Leiter hat die Länge $l=\SI{10}{\centi\meter}$ und
  wird von einem Strom der Stärke $I=\SI{2.5}{\ampere}$ durchflossen. Der
  Leiter befindet sich in einem homogenen Magnetfeld der Flussdichte
  $B=\SI{55}{\milli\tesla}$; es wirkt auf ihn eine magnetische Kraft mit
  dem Betrag $F_{\text{mag}}=\SI{9.72}{\milli\newton}$. Berechnen Sie die
  Weite des Winkels, den Leiter und magnetische Feldlinien einschliessen.
  \Antwortraum{2}
  \begin{solution}
  \[
    \sin\alpha = \frac{F_{\text{mag}}}{I\cdot l\cdot B}
      = \frac{\SI{9.72e-3}{\newton}}{\SI{2.5}{\ampere}\cdot\SI{0.10}{\meter}\cdot\SI{55e-3}{\tesla}}
      \approx 0.707
  \]
  \[
    \alpha = \arcsin(0.707) \approx 45^\circ
  \]
  \end{solution}
\end{parts}
\end{taskbox}
\end{questions}

\begin{questions}
\begin{taskbox}[Aufgabe 4: Bestimmung der Lorentzkraft (Formelumstellung)]
\Schwierigkeit{leicht}
\question Lösen Sie die Formel $F_L=q\cdot v\cdot B\cdot\sin\alpha$ jeweils
nach der gesuchten Grösse auf und berechnen Sie diese.
\begin{parts}
  \part[2] Ein Elektron fliegt mit der Geschwindigkeit
  $v=\SI{1.0e5}{\meter\per\second}$ durch ein homogenes Magnetfeld mit
  der magnetischen Flussdichte $B=\SI{1.2e-2}{\tesla}$. Die Weite des von
  Geschwindigkeits- und Feldvektor eingeschlossenen Winkels beträgt
  $35^\circ$. Berechnen Sie den Betrag der auf das Elektron wirkenden
  Lorentzkraft.
  \Antwortraum{2}
  \begin{solution}
  \[
    F_L = q\cdot v\cdot B\cdot\sin\alpha
      = \SI{1.6e-19}{\coulomb}\cdot\SI{1.0e5}{\meter\per\second}\cdot\SI{1.2e-2}{\tesla}\cdot\sin(35^\circ)
      \approx \SI{1.1e-16}{\newton}
  \]
  \end{solution}
  \part[2] Ein Elektron fliegt durch ein Magnetfeld mit der magnetischen
  Flussdichte $B=\SI{2.15e-4}{\tesla}$ und der Geschwindigkeit
  $v=\SI{4.22e6}{\meter\per\second}$. Auf das Elektron wirkt die
  Lorentzkraft vom Betrag $F_L=\SI{1.26e-16}{\newton}$. Berechnen Sie die
  Weite des von Geschwindigkeits- und Feldvektor eingeschlossenen Winkels.
  \Antwortraum{2}
  \begin{solution}
  \[
    \sin\alpha = \frac{F_L}{q\cdot v\cdot B}
      = \frac{\SI{1.26e-16}{\newton}}{\SI{1.6e-19}{\coulomb}\cdot\SI{4.22e6}{\meter\per\second}\cdot\SI{2.15e-4}{\tesla}}
      \approx 0.870
  \]
  \[
    \alpha = \arcsin(0.870) \approx 60.2^\circ \quad\text{oder}\quad \alpha = 180^\circ-60.2^\circ \approx 119.8^\circ
  \]
  \end{solution}
\end{parts}
\end{taskbox}
\end{questions}

\begin{questions}
\begin{taskbox}[Aufgabe 5: Beschleunigung eines Leiters]
\Schwierigkeit{leicht}
\question Zwei feste Leiter mit dem Abstand $d=\SI{5.0}{\centi\meter}$
bilden eine Metallschiene, auf der ein beweglicher Leiter der Länge
$l=\SI{7.0}{\centi\meter}$ und Masse $m=\SI{100}{\gram}$ liegt.
\begin{center}
  \includegraphics[width=8cm]{Bilder/Beschleunigung_eines_Leiters.png}
\end{center}
{\scriptsize Der bewegliche Leiter liegt schräg auf den beiden festen
Schienen (Abstand $d$) und ragt beidseitig darüber hinaus (Länge $l$).}
\begin{parts}
  \part[2] Berechnen Sie die magnetische Kraft auf den beweglichen
  Leiter, wenn durch den geschlossenen Kreis der Strom $I=\SI{20}{\ampere}$
  fliesst und die magnetische Flussdichte $B=\SI{1.5}{\tesla}$ beträgt.
  Überlegen Sie zuerst genau, durch welchen Teil des beweglichen Leiters
  überhaupt Strom fliesst.
  \Antwortraum{2}
  \begin{solution}
  Der geschlossene Stromkreis hat zwischen den Schienen die Breite $d$,
  nicht die volle Leiterlänge $l$. Der Leiter liegt schräg auf der
  Schiene und ragt beidseitig darüber hinaus, aber nur das Stück der
  Länge $d$ trägt effektiv Strom zwischen den beiden Schienen:
  \[
    F_{\text{mag}} = I\cdot d\cdot B = \SI{20}{\ampere}\cdot\SI{0.050}{\meter}\cdot\SI{1.5}{\tesla} = \SI{1.5}{\newton}
  \]
  \end{solution}
  \part[2] Berechnen Sie die Beschleunigung, die der bewegliche Leiter
  dadurch erfährt.
  \Antwortraum{2}
  \begin{solution}
  Mit dem 2.\ NEWTONschen Gesetz $F=m\cdot a$:
  \[
    a = \frac{F_{\text{mag}}}{m} = \frac{\SI{1.5}{\newton}}{\SI{0.100}{\kilo\gram}} = \SI{15}{\meter\per\second\squared}
  \]
  \end{solution}
\end{parts}
\end{taskbox}
\end{questions}

\begin{questions}
\begin{taskbox}[Aufgabe 6: Flug parallel zum Draht]
\Schwierigkeit{mittelschwer}
\question Ein Elektron bewegt sich mit einer Geschwindigkeit
$v=\SI{5.45e6}{\meter\per\second}$ im Abstand von $r=\SI{8.00}{\centi\meter}$
parallel zu einem langen geraden Draht, der von einem Strom der Stärke
$I=\SI{25.0}{\ampere}$ durchflossen wird.
\begin{center}
\begin{tikzpicture}[>=Latex,scale=0.9]
  \draw[very thick] (0,-1.8) -- (0,2.0);
  \draw[thick,-{Latex[length=3mm]}] (0,-1.5) -- (0,-0.7) node[midway,left,font=\scriptsize]{$I$};
  \node[font=\scriptsize] at (0,2.3) {Draht};
  \draw[dashed,gray] (0,0) circle (0.9);
  \draw[dashed,gray] (0,0) circle (1.6);
  \draw[<->] (0,0) -- (2.3,0) node[midway,above,font=\scriptsize]{$r$};
  \filldraw[blue] (2.3,0) circle (0.06);
  \node[font=\scriptsize,blue] at (2.3,-0.35) {$e^-$};
  \draw[thick,-{Latex[length=3mm]},blue] (2.3,0.2) -- (2.3,1.0) node[midway,right,font=\scriptsize,black]{$v$};
\end{tikzpicture}
\end{center}
{\scriptsize Das Elektron fliegt im Abstand $r$ parallel zum Draht; die
gestrichelten Kreise deuten die (kreisförmigen) Feldlinien des Drahts an.}
\begin{parts}
  \part[2] Berechnen Sie die magnetische Flussdichte $B$, die aufgrund
  des Stroms im Draht an der Flugbahn des Elektrons herrscht.
  \Antwortraum{2}
  \begin{solution}
  \[
    B = \frac{\mu_0}{2\pi}\cdot\frac{I}{r}
      = \SI{2e-7}{\newton\per\ampere\squared}\cdot\frac{\SI{25.0}{\ampere}}{\SI{8.00e-2}{\meter}}
      \approx \SI{6.27e-5}{\tesla}
  \]
  \end{solution}
  \part[3] Berechnen Sie den Betrag der Kraft auf das Elektron. Überlegen
  Sie zuerst, welchen Winkel $\alpha$ die Geschwindigkeit des Elektrons
  mit den (kreisförmigen) Feldlinien des Drahts einschliesst.
  \Antwortraum{3}
  \begin{solution}
  Das Elektron bewegt sich \emph{parallel} zum Draht, seine Geschwindigkeit
  steht deshalb senkrecht zu den (kreisförmigen) Feldlinien des Drahts:
  $\alpha=90^\circ$.
  \[
    F_L = q\cdot v\cdot B = \SI{1.60e-19}{\coulomb}\cdot\SI{5.45e6}{\meter\per\second}\cdot\SI{6.27e-5}{\tesla}
      \approx \SI{5.47e-17}{\newton}
  \]
  \end{solution}
\end{parts}
\end{taskbox}
\end{questions}
